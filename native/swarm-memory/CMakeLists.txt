cmake_minimum_required(VERSION 3.16)
project(swarm_memory LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_path(FUSE_INCLUDE_DIR fuse_lowlevel.h PATH_SUFFIXES fuse3 REQUIRED)
find_library(FUSE_LIBRARY NAMES fuse3 libfuse3.so.3 REQUIRED)
add_executable(swarm-memory main.cpp engine.cpp transport.cpp)
target_include_directories(swarm-memory PRIVATE ${FUSE_INCLUDE_DIR})
target_compile_definitions(swarm-memory PRIVATE _FILE_OFFSET_BITS=64)
target_compile_options(swarm-memory PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(swarm-memory PRIVATE ${FUSE_LIBRARY} OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
install(TARGETS swarm-memory RUNTIME DESTINATION bin)
add_executable(swarm-memory-cache-test cache_test.cpp)
target_link_libraries(swarm-memory-cache-test PRIVATE OpenSSL::Crypto Threads::Threads)
